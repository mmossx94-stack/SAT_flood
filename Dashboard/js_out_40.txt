93:       function initDashboard(response) {
94:         function parseCustomDate(dStr) {
95:           if(!dStr) return 0;
96:           let ms = Date.parse(dStr);
97:           if(!isNaN(ms)) return ms;
98:           let parts = dStr.split(' ');
99:           if(parts.length === 2 && parts[0].includes('-')) {
100:              return Date.parse(parts[0] + 'T' + parts[1] + '+07:00') || 0;
101:           }
102:           if(parts.length === 2 && parts[0].includes('/')) {
103:              let dp = parts[0].split('/');
104:              if(dp.length === 3) return Date.parse(`${dp[2]}-${dp[1]}-${dp[0]}T${parts[1]}:00+07:00`) || 0;
105:           }
106:           // Only date dd/MM/yyyy
107:           if(parts.length === 1 && parts[0].includes('/')) {
108:              let dp = parts[0].split('/');
109:              if(dp.length === 3) return Date.parse(`${dp[2]}-${dp[1]}-${dp[0]}T00:00:00+07:00`) || 0;
110:           }
111:           return 0;
112:         }
113: 
114:         function filterToLatestDay(data, dateField) {
115:           if (!data || data.length === 0) return data;
116:           let maxTime = -1;
117:           let maxDateStr = null;
118:           data.forEach(item => {
119:              let ms = parseCustomDate(item[dateField]);
120:              if (ms > maxTime) {
121:                maxTime = ms;
122:                maxDateStr = new Date(ms).toLocaleDateString('th-TH');
123:              }
124:           });
125:           if (maxTime === -1 || !maxDateStr) return data;
126:           return data.filter(item => {
127:              let ms = parseCustomDate(item[dateField]);
128:              if (ms === 0) return false;
129:              return new Date(ms).toLocaleDateString('th-TH') === maxDateStr;
130:           });
131:         }
132: 
133:         let tempFlood = response.floodData || [];
134:         let tempBkk = response.bkkWaterData || [];
135:         let tempThai = response.thaiWaterData || [];
136:         let tempShelter = response.shelterData || [];
137: 
138:         rawFloodData = filterToLatestDay(tempFlood, 'Report_Date');
139:         rawVulData = response.vulData || [];
140:         rawBkkWaterData = filterToLatestDay(tempBkk, 'observed_at_th');
141:         rawThaiWaterData = filterToLatestDay(tempThai, 'observed_at_th');
142:         rawShelterData = filterToLatestDay(tempShelter, 'fetched_at_th');
143: 
144:         populateDropdowns();
145:         applyFilters();
146: 
147:         initMaps();
148:         renderBkkMap(rawBkkWaterData);
149:         drawTrendChartMock();
150:         initNewFilters();
151:         renderThaiMap(rawThaiWaterData);
152:         updateThaiRanking(rawThaiWaterData);
153:         applyShelterFilters(); // This renders shelter map and populates the summary cards/table
154: 
155:         let maxDate = 0;
156:         rawThaiWaterData.forEach(d => { let ms = parseCustomDate(d.fetched_at_th); if(ms > maxDate) maxDate = ms; });
157:         rawBkkWaterData.forEach(d => { let ms = parseCustomDate(d.fetched_at_th); if(ms > maxDate) maxDate = ms; });
158:         rawFloodData.forEach(d => { let ms = parseCustomDate(d.Ingested_At); if(ms > maxDate) maxDate = ms; });
159:         
160:         let updateText = 'ไม่ระบุ';
161:         if (maxDate > 0) {
162:             let d = new Date(maxDate);
163:             updateText = d.toLocaleDateString('th-TH') + ' ' + d.toLocaleTimeString('th-TH');
164:         } else {
165:             updateText = new Date().toLocaleTimeString('th-TH');
166:         }
167: 
168:         document.getElementById('lastUpdated').innerHTML = `<i class="bi bi-check-circle-fill text-success"></i> อัปเดตล่าสุด: ` + updateText;
169:       }

339:       function updateHealthZoneTable(floodData, allVulData) {
340:         // Create Province -> Health Zone mapping from allVulData
341:         let provToHz = {};
342:         allVulData.forEach(item => {
343:           if (item['จังหวัด'] && item['เขตสุขภาพ']) {
344:             provToHz[item['จังหวัด']] = item['เขตสุขภาพ'];
345:           }
346:         });
347: 
348:         // Group data by Health Zone
349:         let hzMap = {};
350:         floodData.forEach(item => {
351:           let p = item.Province;
352:           if (!p) return;
353:           let hz = provToHz[p] || 'ไม่ระบุเขตสุขภาพ';
354:           if (!hzMap[hz]) {
355:             hzMap[hz] = {
356:               provinces: new Set(),
357:               households: 0,
358:               child: 0, preg: 0, elderly: 0, totalVul: 0
359:             };
360:           }
361:           hzMap[hz].provinces.add(p);
362:           hzMap[hz].households += (parseInt(String(item.Affected_Households).replace(/,/g, '')) || 0);
363:         });
364: 
365:         // Add Vulnerable stats to the zones based on affected provinces
366:         allVulData.forEach(item => {
367:           let p = item['จังหวัด'];
368:           let hz = item['เขตสุขภาพ'];
369:           if (p && hz && hzMap[hz] && hzMap[hz].provinces.has(p)) {
370:             hzMap[hz].child += (parseFloat(item['จำนวนเด็ก 0-4 ปีทั้งหมด (คน)']) || 0);
371:             hzMap[hz].preg += (parseFloat(item['จำนวนหญิงตั้งครรภ์ทั้งหมด (คน)']) || 0);
372:             hzMap[hz].elderly += (parseFloat(item['จำนวนผู้สูงอายุ 60 ปีขึ้นไปทั้งหมด (คน)']) || 0);
373:             hzMap[hz].totalVul += (parseFloat(item['รวมทั้งหมด']) || 0);
374:           }
375:         });
376: 
377:         const tbody = document.getElementById('healthZoneTableBody');
378:         tbody.innerHTML = '';
379: 
380:         let hzKeys = Object.keys(hzMap);
381:         if (hzKeys.length === 0) {
382:           tbody.innerHTML = '<tr><td colspan="7" class="text-center text-muted py-3">ไม่พบข้อมูล</td></tr>';
383:           return;
384:         }
385: 
386:         // Sort by Number of Households descending
387:         let sortedHz = hzKeys.map(k => ({ name: k, ...hzMap[k] })).sort((a, b) => b.households - a.households);
388: 
389:         sortedHz.forEach(hz => {
390:           let tr = document.createElement('tr');
391:           tr.innerHTML = `
392:           <td class="fw-semibold text-success">${hz.name}</td>
393:           <td class="text-center">${hz.provinces.size}</td>
394:           <td class="text-end fw-bold text-danger">${hz.households.toLocaleString()}</td>
395:           <td class="text-end" style="background-color: rgba(0,0,0,0.02);">${hz.child.toLocaleString()}</td>
396:           <td class="text-end" style="background-color: rgba(0,0,0,0.02);">${hz.preg.toLocaleString()}</td>
397:           <td class="text-end" style="background-color: rgba(0,0,0,0.02);">${hz.elderly.toLocaleString()}</td>
398:           <td class="text-end fw-bold text-purple" style="background-color: rgba(0,0,0,0.05);">${hz.totalVul.toLocaleString()}</td>
399:         `;
400:           tbody.appendChild(tr);
401:         });
402:       }

769:       function animateValue(id, start, end, duration) {
770:         if (start === end) { document.getElementById(id).innerHTML = end.toLocaleString(); return; }
771:         let range = end - start;
772:         let current = start;
773:         let increment = end > start ? 1 : -1;
774:         let stepTime = Math.abs(Math.floor(duration / range));
775:         if (stepTime < 10) { increment = Math.ceil(range / (duration / 10)); stepTime = 10; }
776: 
777:         let obj = document.getElementById(id);
778:         let timer = setInterval(function () {
779:           current += increment;
780:           if ((increment > 0 && current >= end) || (increment < 0 && current <= end)) {
781:             current = end;
782:             clearInterval(timer);
783:           }
784:           obj.innerHTML = current.toLocaleString();
785:         }, stepTime);
786:       }
