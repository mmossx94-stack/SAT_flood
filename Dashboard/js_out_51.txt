           let isAlmostFullA = !isFullA && (capA > 0 && occA >= capA * 0.8);
           let scoreA = isFullA ? 2 : (isAlmostFullA ? 1 : 0);

           let occB = parseInt(b.occupied) || 0;
           let capB = parseInt(b.capacity) || 0;
           let isFullB = (b.status_source && b.status_source.includes('เต็ม')) || (capB > 0 && occB >= capB);
           let isAlmostFullB = !isFullB && (capB > 0 && occB >= capB * 0.8);
           let scoreB = isFullB ? 2 : (isAlmostFullB ? 1 : 0);
           
           if (scoreA !== scoreB) {
              return scoreB - scoreA;
           }
           return occB - occA;
        });

           let cap = parseInt(d.capacity) || 0;

           let isAlmostFull = !isFull && (cap > 0 && occ >= cap * 0.8);
