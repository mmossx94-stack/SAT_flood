            provToHz[item['จังหวัด']] = item['เขตสุขภาพ'];
          }
        });

        let hzMap = {};

          let p = item.Province;

              provinces: new Set(),
              households: 0,
              child: 0, preg: 0, elderly: 0, totalVul: 0
            };

          if (p && hz && hzMap[hz] && hzMap[hz].provinces.has(p)) {
