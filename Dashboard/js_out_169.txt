      function applyBkkFilters() {
        const dist = document.getElementById('bkkDistrictFilter').value;
        const stat = 'all';

        let filtered = rawBkkWaterData.filter(d => {
          return (dist === 'all' || d.district_or_area === dist) &&
            (stat === 'all' || d.flood_status_source === stat);
        });

        renderBkkMap(filtered);
        updateBkkRanking(filtered);
      }
