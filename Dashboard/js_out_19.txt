function initMaps() {
        // Base Layers for BKK
        const bkkStreet = L.tileLayer('https://mt1.google.com/vt/lyrs=m&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });
        const bkkTerrain = L.tileLayer('https://mt1.google.com/vt/lyrs=p&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });
        const bkkHybrid = L.tileLayer('https://mt1.google.com/vt/lyrs=y&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });

        // 1. BKK Map
        bkkMap = L.map('bkk-map', {
          center: [13.7563, 100.5018],
          zoom: 10,
          layers: [bkkStreet] // Default to Street for clarity
        });

        const bkkBaseMaps = {
          "แผนที่ถนน (ชัดเจน)": bkkStreet,
          "แผนที่ภูมิประเทศ": bkkTerrain,
          "ภาพดาวเทียม": bkkHybrid
        };
        L.control.layers(bkkBaseMaps).addTo(bkkMap);


        // Base Layers for Thai
        const thaiStreet = L.tileLayer('https://mt1.google.com/vt/lyrs=m&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });

        // 2. Thai Map
        thaiMap = L.map('thai-map', {
          center: [13.736717, 100.523186],
          zoom: 5,
          layers: [thaiStreet] // Default to Street for clarity
        });

        const thaiBaseMaps = {
          "แผนที่ถนน (ชัดเจน)": thaiStreet,
          "แผนที่ภูมิประเทศ": thaiTerrain,
          "ภาพดาวเทียม": thaiHybrid
        };
        L.control.layers(thaiBaseMaps).addTo(thaiMap);

        // 3. Shelter Map
        const shelterStreet = L.tileLayer('https://mt1.google.com/vt/lyrs=m&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });
        const shelterTerrain = L.tileLayer('https://mt1.google.com/vt/lyrs=p&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });
        const shelterHybrid = L.tileLayer('https://mt1.google.com/vt/lyrs=y&hl=th&x={x}&y={y}&z={z}', { attribution: '© Google', maxZoom: 20 });

        shelterMap = L.map('shelter-map', {
          center: [13.736717, 100.523186],
          zoom: 6,
          layers: [shelterStreet]
        });

          "แผนที่ถนน (ชัดเจน)": shelterStreet,
          "แผนที่ภูมิประเทศ": shelterTerrain,
          "ภาพดาวเทียม": shelterHybrid
        };
