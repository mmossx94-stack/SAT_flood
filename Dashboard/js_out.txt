      function populateDropdowns() {
        const healthZones = [...new Set(rawVulData.map(item => item['เขตสุขภาพ']))]
          .filter(val => val && !val.includes('รวม'))
          .sort((a, b) => {
            let numA = parseInt(a.replace(/\D/g, '')) || 0;
            let numB = parseInt(b.replace(/\D/g, '')) || 0;
            return numA - numB;
          });

        populateSelect('healthZoneFilter', healthZones, '🏥 ทุกเขตสุขภาพ');
        updateProvinceDropdown();
      }

      function updateProvinceDropdown() {
        const hzFilter = document.getElementById('healthZoneFilter').value;
        let allowedProvinces = new Set();

        if (hzFilter === 'all') {
          rawFloodData.forEach(item => { if (item.Province) allowedProvinces.add(item.Province); });
        } else {
          let floodProvinces = new Set(rawFloodData.map(item => item.Province).filter(Boolean));
          rawVulData.forEach(item => {
            if (item['เขตสุขภาพ'] === hzFilter && item['จังหวัด'] && floodProvinces.has(item['จังหวัด'])) {
              allowedProvinces.add(item['จังหวัด']);
            }
          });
        }

        let provinces = [...allowedProvinces].sort();
        const provSelect = document.getElementById('provinceFilter');
        const currentSelected = provSelect.value;

        provSelect.innerHTML = '';
        populateSelect('provinceFilter', provinces, '📌 ทุกจังหวัด');

        if (provinces.includes(currentSelected)) {
          provSelect.value = currentSelected;
        } else {
          provSelect.value = 'all';
        }
      }

      function onHealthZoneChange() {
        updateProvinceDropdown();
        applyFilters();
      }

      function populateSelect(elementId, options, defaultText) {
        const select = document.getElementById(elementId);
        select.innerHTML = `<option value="all">${defaultText}</option>`;
        options.forEach(opt => {
          const option = document.createElement('option');
          option.value = opt; option.text = opt;
          select.appendChild(option);
        });
      }

      function applyFilters() {
        const hzFilter = document.getElementById('healthZoneFilter').value;
        const province = document.getElementById('provinceFilter').value;

        let provToHz = {};
        rawVulData.forEach(item => {
          if (item['จังหวัด'] && item['เขตสุขภาพ']) {
            provToHz[item['จังหวัด']] = item['เขตสุขภาพ'];
          }
        });

        currentFilteredData = rawFloodData.filter(item => {
          let hz = provToHz[item.Province] || '';
          return (hzFilter === 'all' || hz === hzFilter) &&
            (province === 'all' || item.Province === province);
        });

        const affectedProvinces = new Set(currentFilteredData.map(item => item.Province).filter(Boolean));
        let filteredVulData = rawVulData.filter(item => affectedProvinces.has(item['จังหวัด']));

        updateSummary(currentFilteredData);
        updateTable(currentFilteredData, rawVulData);
        updateExecutiveSummary(currentFilteredData);
        updateHealthZoneTable(currentFilteredData, rawVulData);

        if (isMapLoaded) {
          drawMap(currentFilteredData, hzFilter); // Pass hzFilter to map
          drawVulnerableChart(filteredVulData);
        }
        updateVulnerableTable(filteredVulData);
      }
