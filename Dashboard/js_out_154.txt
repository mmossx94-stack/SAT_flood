      function initDashboard(response) {

        function filterToLatestDay(data, dateField) {
          if (!data || data.length === 0) return data;
          let maxTime = -1;
          let maxDateStr = null;
          data.forEach(item => {
             let ms = parseCustomDate(item[dateField]);
             if (ms > maxTime) {
               maxTime = ms;
               maxDateStr = new Date(ms).toLocaleDateString('th-TH');
             }
          });
          if (maxTime === -1 || !maxDateStr) return data;
          return data.filter(item => {
             let ms = parseCustomDate(item[dateField]);
             if (ms === 0) return false;
             return new Date(ms).toLocaleDateString('th-TH') === maxDateStr;
          });
        }

        let tempBkk = response.bkkWaterData || [];
        let tempThai = response.thaiWaterData || [];
        let tempShelter = response.shelterData || [];

        rawVulData = response.vulData || [];
        rawBkkWaterData = filterToLatestDay(tempBkk, 'observed_at_th');
