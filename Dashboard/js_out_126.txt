        thaiMarkers.forEach(m => thaiMap.removeLayer(m));
        thaiMarkers = [];

        let bounds = [];

        data.forEach(item => {
          let lat = parseFloat(item.latitude || (item.raw_station_json ? JSON.parse(item.raw_station_json).station.tele_station_lat : null));
          let lng = parseFloat(item.longitude || (item.raw_station_json ? JSON.parse(item.raw_station_json).station.tele_station_long : null));

          if (!isNaN(lat) && !isNaN(lng)) {
            let status = item.flood_status_source || 'ไม่ระบุ';
            let color = getStatusColor(status);
            let marker = L.marker([lat, lng], { icon: createMarkerIcon(color) }).addTo(thaiMap);
            thaiMarkers.push(marker);
            bounds.push([lat, lng]);

            marker.on('click', function () {
              showDetailsPanel('thai-details-panel', item, 'thai');
            });
          }
        });

        if (bounds.length > 0) thaiMap.fitBounds(bounds);
        updateDefaultSummary("thai", data);
      }
