      function applyShelterFilters() {
        const dist = document.getElementById('shelterDistrictFilter').value;
        const statusEl = document.getElementById('shelterStatusFilter');
        const statusFilter = statusEl ? statusEl.value : 'all';

        let filtered = rawShelterData.filter(d => {
          let matchDist = (dist === 'all') || d.district === dist;
          let matchStatus = (statusFilter === 'all') || d.status_source === statusFilter;
          return matchDist && matchStatus;
        });

        renderShelterMap(filtered);
        updateShelterSummary(filtered);
      }
